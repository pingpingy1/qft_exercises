\paragraph{Note:}
I generally disdain the use of natural units in undergraduate studies.
For students having gone through a symbolic mess of a calculation,
something as simple as dimensional analysis gives a good sanity check.
And while this course is explicitly designed for graduate students,
I would like to stand by this conviction and perform all calculations
with explicit engineering dimensions and constants.
I believe it is reasonably simple for the grader to convert it into natural units.

\section*{Problem 1}
We use the following definitions:
\[
    A^\mu = (\frac{\phi}{c}, \vec{A}),\quad
    F^{\mu\nu} = \partial^{[\mu} A^{\nu]},\quad
    J^\mu = (-c\rho, \vec{J}),\quad
\]
\[
    E^i = -\partial_i \phi - \partial_t A^i
    = c\left(\left(-\frac{1}{c}\partial_t\right) A^i - \partial_i \left(\frac{\phi}{c}\right)\right)
    = cF^{0i},\quad
\]
\[
    B^i = \partial_j A^k - \partial_k A^j = F^{jk}\quad(\epsilon_{ijk}=+1).
\]
We start with the Lagrangian density:
\[
    \lagrdens = -\frac{1}{4\mu_0} F_{\alpha\beta}F^{\alpha\beta} + J_\alpha A^\alpha.
\]
We find the following partial derivative relations:
\[
    \frac{\partial(\partial_\alpha A_\beta)}{\partial(\partial_\mu A_\nu)}
    = \delta^\mu_\alpha \delta^\nu_\beta
    \Rightarrow
    \frac{\partial(\partial^\alpha A^\beta)}{\partial(\partial_\mu A_\nu)}
    = \eta^{\mu\alpha} \eta^{\nu\beta}
\]
\[
    \Rightarrow
    \frac{\partial}{\partial(\partial_\mu A_\nu)} (F_{\alpha\beta}F^{\alpha\beta})
    = (\delta^\mu_\alpha \delta^\nu_\beta - \delta^\mu_\beta \delta^\nu_\alpha) F^{\alpha\beta}
      + F_{\alpha\beta} (\eta^{\mu\alpha}\eta^{\nu\beta} - \eta^{\mu\beta}\eta^{\nu\alpha})
    = 4F^{\mu\nu}
\]
Thus, the Euler-Lagrange equations are given by
\[
    0
    = \partial_\mu \left(\frac{\partial\lagrdens}{\partial(\partial_\mu A_\nu)}\right)
      - \frac{\partial\lagrdens}{\partial A_\nu}
    = \partial_\mu \left(-\frac{1}{\mu_0} F^{\mu\nu}\right) - J^\nu
\]
\[
    \therefore \partial_\mu F^{\mu\nu} = \mu_0 J^\nu
\]
\begin{enumerate}
    \item $\nu = 0$:
        $\partial_\mu F^{\mu 0} = -\frac{1}{c} \nablavec \cdot \vec{E}
        \Rightarrow \nablavec\cdot\vec{E} = \frac{\rho}{\epsilon_0}$

    \item $\nu = 1$:
        $\partial_\mu F^{\mu 1}
        = \left(\frac{1}{c}\partial_t\right)\left(\frac{E_x}{c}\right)
          + \partial_y (-B^z) + \partial_z B^y
        = \left(\frac{1}{c^2}\partial_t\vec{E} - \nablavec\times\vec{B}\right)^x$. \\
        Thus, generally, $\nablavec\times\vec{B} = \mu_0\vec{J} + \frac{1}{c^2}\partial_t\vec{E}$.
\end{enumerate}
Furthermore, the Bianchi identity can be expressed as follows:
\begin{enumerate}
    \item
        $\partial_{[0}F_{12]}
        = \frac{1}{c}\partial_t B^z
          + \partial_x \left(+\frac{E^y}{c}\right)
          + \partial_y \left(-\frac{E^x}{c}\right)
        = \frac{1}{c}\left(\partial_t \vec{B} + \nablavec\times\vec{E}\right)^z$. \\
        Thus, generally, $\nablavec\times\vec{E} = -\partial_t\vec{B}$.

    \item
        $\partial_{[1}F_{23]}
        = \partial_x B^x + \partial_y B^y + \partial_z B^z
        = \nablavec\cdot\vec{B} = 0$.
\end{enumerate}
Thus, we see that the Euler-Lagrange equations as well as the Bianchi identity
successfully reproduce the conventional Maxwell equations.\qed

\section*{Problem 2}
From the special-relativistic identity $E^2 = \sqrt{\vec{p}^2c^2 + m^2c^4}$,
we attempt to define the Hamiltonian operator as:
\[
    H \overset{?}{=} \sqrt{-\hbar^2c^2\nablavec^2 + m^2c^4}
    = mc^2\sqrt{1-\left(\frac{\hbar}{mc}\nablavec\right)^2}.
\]
To transform this into a kernel function, i.e., find the function $K(\vec{x}, \vec{y})$ such that
\[
    (H\psi)(\vec{x}) = \int d^3\vec{y} K(\vec{x}, \vec{y}) \psi(\vec{y}),
\]
we utilize the Fourier transform:
\[
    \psi(\vec{x}) = \int d^3\vec{k} \psi_{\vec{k}} e^{i\vec{k}\cdot\vec{x}},\quad
    \psi_{\vec{k}} = \int \frac{d^3\vec{x}}{(2\pi)^3} \psi(\vec{x}) e^{-i\vec{k}\cdot\vec{x}}.
\]
Substitution then yields:
\begin{align*}
    (H\psi)(\vec{x})
    &= mc^2 \int d^3\vec{k} \psi_{\vec{k}} \sqrt{1 - \left(
        \frac{\hbar}{mc}\nablavec
    \right)^2} e^{i\vec{k}\cdot\vec{x}} \\
    &= mc^2 \int d^3\vec{k} \psi_{\vec{k}} \sqrt{1-\left(\frac{\hbar\vec{k}}{mc}\right)^2} e^{i\vec{k}\cdot\vec{x}} \\
    &= mc^2\int d^3\vec{k} \left(
        \int\frac{d^3\vec{y}}{(2\pi)^3} \psi(\vec{y}) e^{-i\vec{k}\cdot\vec{y}}
    \right) \sqrt{1-\left(\frac{\hbar\vec{k}}{mc}\right)^2} e^{i\vec{k}\cdot\vec{x}} \\
    &= \int d^3\vec{y} \left(
        mc^2 \int\frac{d^3\vec{k}}{(2\pi)^3}
        \sqrt{1-\left(\frac{\hbar\vec{k}}{mc}\right)^2}
        e^{i\vec{k}\cdot\left(\vec{x}-\vec{y}\right)}
    \right) \psi(\vec{y})
\end{align*}
Thus, defining $\vec{r} := \vec{x} - \vec{y}$, we find that
\begin{align*}
    K(\vec{x}, \vec{y})
    &= mc^2 \int\frac{d^3\vec{k}}{(2\pi)^3}
        \sqrt{1 - \left(\frac{\hbar\vec{k}}{mc}\right)^2} e^{i\vec{k}\cdot\vec{r}} \\
    &= \frac{m^4c^5}{8\pi^3\hbar^3}
        \int d^3\vec{\kappa} \sqrt{1+\vec{\kappa}^2} e^{i\vec{\kappa}\cdot\vec{\rho}}
    & \left(
        \vec{\kappa} := \frac{\hbar\vec{k}}{mc},\quad
        \vec{\rho} := \frac{mc\vec{r}}{\hbar}
    \right) \\
    &= \frac{m^4c^5}{8\pi^3\hbar^3} \int_0^\infty d\kappa \kappa^2\sqrt{1+\kappa^2}
        \int_{-1}^1 d(\cos\theta) e^{i\kappa\rho\cos\theta}
        \int_0^{2\pi} d\phi \\
    &= \frac{m^4c^5}{2\pi^2\hbar^3\rho}
        \int_0^\infty d\kappa \kappa\sqrt{1+\kappa^2}\sin\kappa\rho.
\end{align*}
Despite the obvious problem of convergence,
we nevertheless attempt to sidestep the issue using appropriate Feynmann tricks.
\begin{align*}
    K(\vec{x}, \vec{y})
    &= \frac{m^4c^5}{2\pi^2\hbar^3\rho}
        \int_0^\infty d\kappa \left(
            \frac{\kappa^3}{\sqrt{1+\kappa^2}}
            + \frac{\kappa}{\sqrt{1+\kappa^2}}
        \right) \sin\kappa\rho \\
    &= \frac{m^4c^5}{2\pi^2\hbar^3\rho} \left(
        \frac{d^3}{d\rho^3} - \frac{d}{d\rho}
    \right) \int_0^\infty d\kappa \frac{\sin\kappa\rho}{\sqrt{1+\kappa^2}} \\
    &= \frac{m^4c^5}{2\pi^2\hbar^3\rho} \left( K_0'''(\rho) - K_0'(\rho) \right),
\end{align*}
where $K_0(x)$ is the modified Bessel function of the second kind of order $0$.
The last expression can be simplified using well-known recurrence relations:
\[
    K_0'(x) = -K_1(x),\quad
    K_1'(x) = -K_0(x) - \frac{1}{x} K_1(x),\quad
    K_2(x) = K_0(x) + \frac{2}{x} K_1(x).
\]
\[
    \Rightarrow K_0'''(x) - K_0'(x) = -\frac{1}{x}K_2(x)
\]
\[
    \therefore K(\vec{x}, \vec{y})
    = -\frac{m^4c^5}{2\pi^2\hbar^3} \frac{K_2(\rho)}{\rho^2}
    = -\frac{m^2c^3}{2\pi^2\hbar} \frac{K_2\left(\frac{mc}{\hbar}r\right)}{r^2}
\]
Notice the blatant nonlocality: $K(\vec{x}, \vec{y})$ has support arbitrarily far away from $\vec{x} = \vec{y}$!
In fact, the asymptotic behavior $K_\alpha(x) \overset{x\rightarrow\infty}{\sim}
\sqrt{\frac{\pi}{2x}}e^{-x}$,
we find that
\[
    K(\vec{x}, \vec{y}) \overset{r\rightarrow\infty}{\sim}
    -\sqrt{\frac{m^3c^5}{8\pi^3\hbar}} r^{-2} e^{-\frac{mc}{\hbar}r}.
\]
